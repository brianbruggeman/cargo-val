   150	        bytes = &bytes[written as usize..];
   151	    }
   152	    Ok(())
   153	}

   155	fn phase_decimal(mut value: u64, output: &mut [u8], position: &mut usize) {
   156	    let mut digits = [0_u8; 20];
   157	    let mut count = 0;
   158	    loop {
   159	        digits[count] = b'0' + (value % 10) as u8;
   160	        count += 1;
   161	        value /= 10;
   162	        if value == 0 {
   163	            break;
   164	        }
   165	    }
   166	    while count != 0 {
   167	        count -= 1;
   168	        output[*position] = digits[count];
   169	        *position += 1;
   170	    }
   171	}

   173	fn phase_event(label: &[u8], process_id: libc::pid_t, count: usize) -> Result<(), RunError> {
   174	    let mut now = libc::timespec { tv_sec: 0, tv_nsec: 0 };
   175	    if unsafe { libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut now) } != 0 {
   176	        return Err(RunError::Reactor);
   177	    }
   178	    let nanos = now.tv_sec as u64 * 1_000_000_000 + now.tv_nsec as u64;
   179	    let mut line = [0_u8; 128];
   180	    let mut offset = 0;
   181	    for fragment in [b"C54PHASE ".as_slice(), label, b" "] {
   182	        line[offset..offset + fragment.len()].copy_from_slice(fragment);
   183	        offset += fragment.len();
   184	    }
   185	    phase_decimal(nanos, &mut line, &mut offset);
   186	    line[offset] = b' ';
   187	    offset += 1;
   188	    phase_decimal(process_id as u64, &mut line, &mut offset);
   189	    line[offset] = b' ';
   190	    offset += 1;
   191	    phase_decimal(count as u64, &mut line, &mut offset);
   192	    line[offset] = b'\n';
   193	    write_fd(2, &line[..offset + 1])
   194	}

   630	pub fn run_unix(parser: &CliParser<'_>) -> Result<RunSummary, RunError> {
   631	    phase_event(b"run_start", 0, 0)?;
   632	    if parser.has_invalid_manifest_path()
   633	        || parser.has_invalid_cargo_option_value()
   634	        || parser.has_invalid_positional_argument()
   635	        || parser.has_argument_storage_overflow()
   636	    {
   637	        return Err(RunError::Arguments);
   638	    }
   639	    let environment = unsafe { UnixEnvironment::current() };
   640	    let settings = runtime_settings(&environment)?;
   641	    let mut inherited = [c""; ENV_CAPACITY];
   642	    let inherited_count = environment
   643	        .collect(&mut inherited)
   644	        .map_err(|_| RunError::Environment)?;
   645	    let inherited = &inherited[..inherited_count];
   646	    let mut envp = [core::ptr::null(); ENV_CAPACITY + 1];
   647	    let envp_count = environment_pointers(inherited, &mut envp)?;
   648	    let envp = &envp[..envp_count];
   649	    let mut cwd_bytes = [0; PATH_CAPACITY];
   650	    let cwd = working_directory(&mut cwd_bytes)?;
   651	    let filter = parser
   652	        .filter()
   653	        .map(core::str::from_utf8)
   654	        .transpose()
   655	        .map_err(|_| RunError::Arguments)?
   656	        .unwrap_or("");
   657	    let mut argv = [core::ptr::null(); 64];
   658	    let args = parser.cargo_arguments();
   659	    if args.len() + 5 > argv.len() {
   660	        return Err(RunError::Arguments);
   661	    }
   662	    argv[0] = c"cargo".as_ptr();
   663	    argv[1] = c"test".as_ptr();
   664	    argv[2] = c"--no-run".as_ptr();
   665	    argv[3] = c"--message-format=json".as_ptr();
   666	    for (index, argument) in args.iter().enumerate() {
   667	        if argument.contains(&0) {
   668	            return Err(RunError::Arguments);
   669	        }
   670	        argv[4 + index] = argument.as_ptr().cast();
   671	    }
   672	    let cargo_child = spawn_raw_unix(c"cargo", &argv[..args.len() + 5], envp, None, true)
   673	        .map_err(|_| RunError::Spawn)?;
   674	    phase_event(b"cargo_spawn", cargo_child.process_id(), 0)?;
   675	    let cargo_process_id = cargo_child.process_id();
   676	    let port = NativePort::new().map_err(|_| RunError::Reactor)?;
   677	    let (mut cargo_stdout, mut cargo_stderr, mut cargo_process) = reader_pair(&port, cargo_child)?;
   678	    let mut cargo_source_bytes = [0; SOURCE_CAPACITY];
   679	    let mut cargo_frame_bytes = [0; FRAME_CAPACITY];
   680	    let mut cargo_scratch = [0; PATH_CAPACITY];
   681	    let mut source =
   682	        CargoSource::new(&mut cargo_source_bytes).map_err(|_| RunError::CargoStream)?;
   683	    let mut framer = ArtifactFramer::new(&mut cargo_frame_bytes, FRAME_CAPACITY - 1)
   684	        .map_err(|_| RunError::CargoStream)?;
   685	    let mut sessions = SessionSlots::<RunningUnixChild<NAME_CAPACITY>, LIVE_CAPACITY>::new();
   686	    let mut ledger = OutcomeLedger::new(&mut sessions).map_err(|_| RunError::Outcome)?;
   687	    let mut watches: [Option<TestWatch>; LIVE_CAPACITY] = [const { None }; LIVE_CAPACITY];
   688	    let mut events = [NativeEvent { token: 0 }; 64];
   689	    let mut context = Context::from_waker(Waker::noop());
   690	    let mut cargo_stdout_eof = false;
   691	    let mut cargo_stderr_eof = false;
   692	    let mut cargo_exit_status = cargo_process.reap().map_err(|_| RunError::Reactor)?;
   693	    if cargo_exit_status.is_some() {
   694	        phase_event(b"cargo_exit", cargo_process_id, 0)?;
   695	    }
   696	    while !cargo_stdout_eof
   697	        || !cargo_stderr_eof
   698	        || cargo_exit_status.is_none()
   699	        || ledger.unaccounted() != 0
   700	    {
   735	                    Poll::Ready(Ok(CargoSourceStep::Eof)) => {
   736	                        cargo_stdout_eof = true;
   737	                        break;
   738	                    }
   739	                    Poll::Ready(Err(_)) => return Err(RunError::CargoStream),
   740	                    Poll::Pending => break,
   741	                }
   742	            }
   743	        }
   744	        if !cargo_stderr_eof {
   745	            cargo_stderr_eof = drain_reader(&mut cargo_stderr, 2)?;
   746	        }
   747	        if let Some(status) = cargo_process.reap().map_err(|_| RunError::Reactor)? {
   748	            if cargo_exit_status.is_none() {
   749	                phase_event(b"cargo_exit", cargo_process_id, 0)?;
   750	            }
   751	            cargo_exit_status = Some(status);
   752	        }
   753	        if cargo_stdout_eof
   754	            && cargo_stderr_eof
   755	            && cargo_exit_status.is_some()
   756	            && ledger.unaccounted() == 0
   757	        {
   758	            break;
   759	        }
   760	        #[cfg(target_os = "macos")]
   761	        let ready = port.wait_for(&mut events, 100).map_err(|_| RunError::Reactor)?;
   762	        #[cfg(target_os = "linux")]
   763	        let ready = port.wait(&mut events).map_err(|_| RunError::Reactor)?;
   764	        if ready == 0 {
   765	            reap_unreported_children(&port, &mut ledger, &mut watches, settings.color)?;
   766	        }
   767	        for event in events.iter().take(ready) {
   768	            if event.token == cargo_stdout.token() {
   769	                cargo_stdout.wake_read_ready();
   770	            } else if event.token == cargo_stderr.token() {
   771	                cargo_stderr.wake_read_ready();
   772	            } else if event.token == cargo_process.token() {
   773	                if let Some(status) = cargo_process.reap().map_err(|_| RunError::Reactor)? {
   774	                    if cargo_exit_status.is_none() {
   775	                        phase_event(b"cargo_exit", cargo_process_id, 0)?;
   776	                    }
   777	                    cargo_exit_status = Some(status);
   778	                }
   779	            } else {
   780	                test_event(
   781	                    &port,
   782	                    event.token,
   783	                    &mut ledger,
   784	                    &mut watches,
   785	                    settings.color,
   786	                )?;
   787	            }
   788	        }
   789	    }
   790	    if cargo_exit_status.is_some_and(|status| status.code != Some(0)) {
   791	        return Err(RunError::CargoFailed);
   792	    }
   793	    phase_event(b"cargo_done", cargo_process_id, ledger.snapshot().discovered)?;
   794	    let counts = ledger.snapshot();
   795	    phase_event(b"run_done", 0, counts.passed + counts.failed)?;
   796	    Ok(RunSummary {
   797	        passed: counts.passed,
   798	        failed: counts.failed,
   799	    })
