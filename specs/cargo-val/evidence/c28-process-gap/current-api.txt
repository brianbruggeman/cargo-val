     1	//! libc-based process spawn primitive.
     2	//!
     3	//! Consumes a tier-1 [`CommandDescriptor`] descriptor and produces a
     4	//! [`Child`] containing the child's pid plus parent-side
     5	//! file descriptors for every [`Stdio::Piped`] slot.
     6	//!
     7	//! # What it honors
     8	//!
     9	//! Every field of [`CommandDescriptor`]:
    10	//! - `program` + `args` → `execvp` argv
    11	//! - `cwd` → `chdir(2)` in the child before exec
    12	//! - `env` → built into a `*const *const c_char` envp **in the
    13	//!   parent** (allocation-safe context), then handed to the child
    14	//!   via a single `environ` pointer swap. The child performs **no**
    15	//!   `setenv` / `unsetenv` / `clearenv` calls — pure data pass-
    16	//!   through. `execvp` uses the swapped `environ` for both the
    17	//!   child's env AND its `PATH` search.
    18	//! - `umask` → `umask(2)` in the child before exec
    19	//! - `controlling_tty` → `setsid()` + `ioctl(TIOCSCTTY, 0)` on fd 0
    20	//! - per-slot [`Stdio`] wiring:
    21	//!   - [`Stdio::Inherit`] — child inherits parent's fd at that index
    22	//!   - [`Stdio::Null`] — open `/dev/null` in the child, dup2 onto the slot
    23	//!   - [`Stdio::Fd`]`(n)` — child dup2s `n` onto the slot (caller retains
    24	//!     ownership of `n`)
    25	//!   - [`Stdio::Piped`] — `pipe(2)` allocated in the parent before fork;
    26	//!     child dup2s its end, parent keeps the other in [`Child`]
    27	//!
    28	//! # Hazard
    29	//!
    30	//! `fork(2)` from a multi-threaded parent is unsafe (held mutexes
    31	//! survive into the child with no thread to release them). Direct
    32	//! callers of `spawn` should route through
    33	//! [`ForkServer`](super::fork_server::ForkServer) when the parent
    34	//! is multi-threaded. The fork-server is single-threaded by
    35	//! construction so its `spawn` is safe; multi-threaded direct
    36	//! callers risk deadlocks.

   158	fn spawn_request_pump(write_end: OwnedFd, request: Request<Bytes>) {
   159	    let raw_fd = write_end.into_raw_fd();
   160	    thread::spawn(move || {
   161	        let mut writer = unsafe { std::fs::File::from_raw_fd(raw_fd) };
   162	        let mut stream = request.into_chunk_stream();
   163	        block_on(async {
   164	            while let Some(result) = stream.next().await {
   165	                let Ok(bytes) = result else { break };
   166	                if writer.write_all(&bytes).is_err() {
   167	                    break;
   168	                }
   169	            }
   170	        });
   171	    });

     2	//! typed dispatch chain attached.
     3	//!
     4	//! # Composition
     5	//!
     6	//! `CommandPipe::call` composes:
     7	//!
     8	//! 1. [`spawn_and_dispatch`] — fork+exec the child with the
     9	//!    dispatch chain wired to `extra_fd[7]` + `PROXIMA_DISPATCH_FD`
    10	//!    in the child env. A sibling thread drives the chain over the
    11	//!    parent end of a socketpair.
    12	//! 2. [`FdPairPipe`] — wrap the (parent_write_fd, parent_read_fd)
    13	//!    pair as a Pipe; its `call(request)` does the byte shuttle.
    14	//! 3. Background reap — when the response body fully drains, a
    15	//!    detached thread runs [`DispatchedChild::wait`] which joins
    16	//!    the dispatch thread and `waitpid`s the child.
   165	    fn call(
   166	        &self,
   167	        request: Request<Bytes>,
   168	    ) -> impl Future<Output = Result<Response<Bytes>, ProximaError>> + Send {
   169	        let mut command = self.command.clone();
   170	        command.stdin(Stdio::Piped).stdout(Stdio::Piped);
   171	        let shim_result = if self.libc_shim {
   172	            attach_libc_shim_env(&mut command)
   173	        } else {
   174	            Ok(())
   175	        };
   176	        let chain = self.chain.clone();
   177
   178	        async move {
   179	            shim_result?;
   180	            let mut dispatched = spawn_and_dispatch(&command, chain)?;
   181
   182	            let input_fd = dispatched
   183	                .child
   184	                .stdin
   185	                .take()
   186	                .ok_or_else(|| ProximaError::Body("spawn returned no Piped input fd".into()))?;
   187	            let output_fd =
   188	                dispatched.child.stdout.take().ok_or_else(|| {
   189	                    ProximaError::Body("spawn returned no Piped output fd".into())
   190	                })?;

   233	                        thread::spawn(move || {
   234	                            let _ = dispatched.wait();
   235	                        });
   236	                    }

   493	    /// Mirrors [`std::process::Command::spawn`] in shape. Lowers
   494	    /// to the descriptor, applies [`SpawnOptions`], fork+execs.
   495	    /// Returns a [`Child`] with the parent-side fds for any
   496	    /// `Stdio::Piped` slots — no chain/dispatch wiring here
   497	    /// (use [`Pipe::call`](proxima_primitives::pipe::Pipe::call) for that — it handles the dispatch
   498	    /// thread lifecycle alongside the byte shuttle).
   499	    pub fn spawn(&mut self) -> Result<Child, ProximaError> {
   500	        let descriptor = self.to_descriptor()?;
   501	        spawn(&descriptor, self.spawn_options(None))
   502	    }

   504	    /// Mirrors [`std::process::Command::output`] in shape. Spawns
   505	    /// the child with `stdout` + `stderr` forced to
   506	    /// [`Stdio::Piped`], collects both, waits for exit, returns
   507	    /// status + buffered output. `stdin` stays whatever the
   508	    /// caller configured (default `Inherit`).
   509	    ///
   510	    /// Drop-in alignment with std uses `io::Result<Output>`; we
   511	    /// return `Result<Output, ProximaError>` because the rest of
   512	    /// the crate uses `ProximaError` — same shape, different
   513	    /// error type.
   514	    pub fn output(&mut self) -> Result<Output, ProximaError> {
   515	        let original_stdout = self.stdout;
   516	        let original_stderr = self.stderr;
   517	        self.stdout = Stdio::Piped;
   518	        self.stderr = Stdio::Piped;
   519	        let result = self.spawn_and_collect();
   520	        self.stdout = original_stdout;
   521	        self.stderr = original_stderr;
   522	        result
   523	    }

   533	    fn spawn_and_collect(&mut self) -> Result<Output, ProximaError> {
   534	        let mut child = self.spawn()?;
   535	        let stdout_fd = child
   536	            .stdout
   537	            .take()
   538	            .ok_or_else(|| ProximaError::Body("output requires piped stdout".into()))?;
   539	        let stderr_fd = child
   540	            .stderr
   541	            .take()
   542	            .ok_or_else(|| ProximaError::Body("output requires piped stderr".into()))?;
   543	        let stdout_bytes = drain_fd(stdout_fd);
   544	        let stderr_bytes = drain_fd(stderr_fd);
   545	        let status = wait_child(&mut child)?;
   546	        Ok(Output {
   547	            status,
   548	            stdout: stdout_bytes,
   549	            stderr: stderr_bytes,
   550	        })
